            when {
                expression { return params.TEST_PLAN?.trim() }

                withCredentials([usernamePassword(credentialsId: 'xray-cloud-api',
                                                   usernameVariable: 'XRAY_CLIENT_ID',
                                                   passwordVariable: 'XRAY_CLIENT_SECRET')]) {
                    sh '''
                        # TODO: confirm real cucumber-json output path and Xray Cloud import endpoint.
                        AUTH_TOKEN=$(curl -s -H "Content-Type: application/json" \\
                            -X POST -d "{\\"client_id\\": \\"${XRAY_CLIENT_ID}\\", \\"client_secret\\": \\"${XRAY_CLIENT_SECRET}\\"}" \\
                            "${xray_url}/api/v2/authenticate" | tr -d '"')

                        if [ -f test-results/cucumber-report.json ]; then
                            curl -s -H "Content-Type: application/json" \\
                                -H "Authorization: Bearer ${AUTH_TOKEN}" \\
                                -X POST "${xray_url}/api/v2/import/execution/cucumber?testPlanKey=${TEST_PLAN_1}" \\
                                --data @test-results/cucumber-report.json
                        else
                            echo "No cucumber-report.json found, skipping Xray import"
                        fi
                    '''
                }
            }
        }
    }
