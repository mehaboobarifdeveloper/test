stage('Publish to Xray') {
    when {
        expression { return params.TEST_PLAN?.trim() }
    }
    steps {
        withCredentials([string(credentialsId: 'test1_jiratest2clientid', variable: 'jira_test2client_id'),
                          string(credentialsId: 'test1_jiratest2clientsecret', variable: 'jira_test2client_secret')]) {
            sh '''
                AUTH_TOKEN=$(curl -s -H "Content-Type: application/json" \\
                    -X POST -d "{\\"client_id\\": \\"${jira_test2client_id}\\", \\"client_secret\\": \\"${jira_test2client_secret}\\"}" \\
                    "${xray_url}/api/v2/authenticate" | tr -d '"')

                if [ -f reports/cucumber-report.json ]; then
                    cat > reports/xray-info.json <<EOF
{
  "fields": {
    "project": { "key": "Project_Key" },
    "summary": "Automated run - Build ${BUILD_NUMBER} (${EXEC_ENV})",
    "issuetype": { "name": "Test Execution" }
  },
  "xrayFields": {
    "testPlanKey": "${TEST_PLAN_1}"
  }
}
EOF
                    curl -s -H "Content-Type: multipart/form-data" \\
                        -H "Authorization: Bearer ${AUTH_TOKEN}" \\
                        -X POST \\
                        -F info=@reports/xray-info.json \\
                        -F results=@reports/cucumber-report.json \\
                        "${xray_url}/api/v2/import/execution/cucumber/multipart"
                else
                    echo "No cucumber-report.json found, skipping Xray import"
                fi
            '''
        }
    }
}
